    66→export function AdminStoresTab() {

    68→  const [loading, setLoading] = useState(true);
    69→  const [dialogOpen, setDialogOpen] = useState(false);

    72→  const [deleteTarget, setDeleteTarget] = useState<Store | null>(null);
    73→  const [menuStore, setMenuStore] = useState<Store | null>(null);
    74→  const [filter, setFilter] = useState<"ALL" | StoreType>("ALL");
    75→  const [search, setSearch] = useState("");
    76→
    77→  const load = useCallback(() => {
    78→    setLoading(true);
    79→    fetch("/api/stores")
    80→      .then((r) => r.json())

    94→        (!search.trim() || s.name.includes(search.trim()) || s.category.includes(search.trim()))

    96→  }, [stores, filter, search]);
    97→
    98→  function openCreate() {

   100→    setDialogOpen(true);

   102→

   123→
   124→  async function save() {

   126→      toast({ title: "اسم المتجر مطلوب", variant: "destructive" });
   127→      return;
   128→    }
   129→    setSaving(true);
   130→    const payload = {
   131→      name: form.name.trim(),
   132→      description: form.description.trim() || null,
   133→      type: form.type,
   134→      category: form.category,
   135→      imageUrl: form.imageUrl.trim() || null,
   136→      deliveryFee: Number(form.deliveryFee) || 0,
   137→      deliveryTime: form.deliveryTime.trim() || "30-45 دقيقة",
   138→      minOrder: Number(form.minOrder) || 0,
   139→      phone: form.phone.trim() || null,
   140→      address: form.address.trim() || null,
   141→      lat: Number(form.lat) || 34.1329,
   142→      lng: Number(form.lng) || 42.3775,
   143→      isOpen: form.isOpen,
   144→      isFeatured: form.isFeatured,
   145→    };

   148→        method: form.id ? "PATCH" : "POST",
   149→        headers: { "Content-Type": "application/json" },
   150→        body: JSON.stringify(payload),
   151→      });
   152→      const data = await res.json();
   153→      if (!res.ok) throw new Error(data.error);

   205→      {/* القائمة */}
   206→      {loading ? (

   208→      ) : filtered.length === 0 ? (
   209→        <div className="bg-white rounded-2xl p-10 text-center border border-dashed border-[#FFAA00]">

   219→                <img src={s.imageUrl || (s.type === "MARKET" ? "/placeholder-market.svg" : s.type === "STORE" ? "/placeholder-store.svg" : "/placeholder-restaurant.svg")} alt={s.name} className="w-full h-full object-cover" />

   221→              <div className="flex-1 min-w-0">
   222→                <div className="flex items-center gap-1.5 flex-wrap">
   223→                  <h4 className="font-black text-sm truncate">{s.name}</h4>

   237→                    variant="outline"

   239→                    onClick={() => toggleOpen(s)}
   240→                  >
   241→                    <Power className="h-3 w-3 me-1" /> {s.isOpen ? "مفتوح" : "مغلق"}
   242→                  </Button>
   243→                  <Button size="sm" variant="outline" className="h-7 w-7 rounded-full p-0 text-red-500 hover:bg-red-50" onClick={() => setDeleteTarget(s)} aria-label={`حذف ${s.name}`}>
   244→                    <Trash2 className="h-3.5 w-3.5" />
   245→                  </Button>
   246→                </div>
   247→              </div>
   248→            </div>
   249→          ))}
   250→        </div>
   251→      )}
   252→
   253→      {/* نافذة الإضافة/التعديل */}
   254→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>

   256→          <DialogHeader>

   259→          </DialogHeader>
   260→          <div className="space-y-3.5">
   261→            <div className="space-y-1.5">
   262→              <Label>نوع المتجر *</Label>
   263→              <div className="grid grid-cols-3 gap-2">
   264→                {TYPE_OPTIONS.map((t) => (
   265→                  <button
   266→                    key={t.value}
   267→                    type="button"
   268→                    onClick={() => { setForm((f) => ({ ...f, type: t.value, category: CATEGORY_PRESETS[t.value][0] })); }}
   269→                    className={`rounded-xl border p-2.5 text-center text-xs font-bold transition ${form.type === t.value ? "border-[#FF6B00] bg-[#FFF3E6]" : "border-border hover:border-[#FFAA00]"}`}
   270→                  >
   271→                    <div className="text-lg">{t.icon}</div>

   278→              <div className="space-y-1.5">

   291→                </Select>

   293→            </div>
   294→            <div className="space-y-1.5">
   295→              <Label htmlFor="st-desc">الوصف</Label>
   296→              <Textarea id="st-desc" value={form.description} onChange={(e) => setForm((f) => ({ ...f, description: e.target.value }))} placeholder="وصف قصير يظهر للعملاء..." className="bg-[#F8F9FA] min-h-16" />
   297→            </div>
   298→            <div className="space-y-1.5">
   299→              <Label htmlFor="st-img">رابط صورة المتجر</Label>

   301→              {form.imageUrl.trim() && (
   302→                <div className="h-24 w-40 rounded-xl overflow-hidden border">
   303→                  <img src={form.imageUrl} alt="معاينة" className="w-full h-full object-cover" />
   304→                </div>
   305→              )}
   306→            </div>
   307→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   308→              <div className="space-y-1.5">
   309→                <Label htmlFor="st-fee">أجرة التوصيل (د.ع)</Label>
   310→                <Input id="st-fee" type="number" value={form.deliveryFee} onChange={(e) => setForm((f) => ({ ...f, deliveryFee: e.target.value }))} className="bg-[#F8F9FA]" />
   311→              </div>
   312→              <div className="space-y-1.5">
   313→                <Label htmlFor="st-min">أدنى طلب (د.ع)</Label>
   314→                <Input id="st-min" type="number" value={form.minOrder} onChange={(e) => setForm((f) => ({ ...f, minOrder: e.target.value }))} className="bg-[#F8F9FA]" />
   315→              </div>
   316→              <div className="space-y-1.5">
   317→                <Label htmlFor="st-time">وقت التوصيل</Label>

   326→              <div className="space-y-1.5">
   327→                <Label htmlFor="st-addr">العنوان</Label>
   328→                <Input id="st-addr" value={form.address} onChange={(e) => setForm((f) => ({ ...f, address: e.target.value }))} placeholder="حديثة – حي المعلمين" className="bg-[#F8F9FA]" />
   329→              </div>
   330→            </div>
   331→            <div className="grid grid-cols-2 gap-3">
   332→              <div className="space-y-1.5">
   333→                <Label htmlFor="st-lat">خط العرض (Lat) — خريطة حديثة</Label>

   335→              </div>
   336→              <div className="space-y-1.5">
   337→                <Label htmlFor="st-lng">خط الطول (Lng) — خريطة حديثة</Label>
   338→                <Input id="st-lng" type="number" step="0.0001" value={form.lng} onChange={(e) => setForm((f) => ({ ...f, lng: e.target.value }))} dir="ltr" className="bg-[#F8F9FA] text-left text-xs" />

   345→            <div className="flex items-center justify-between rounded-xl border p-3">
   346→              <Label htmlFor="st-feat" className="cursor-pointer">عرضه ضمن «الأكثر رواجاً»</Label>
   347→              <Switch id="st-feat" checked={form.isFeatured} onCheckedChange={(v) => setForm((f) => ({ ...f, isFeatured: v }))} />
   348→            </div>

   350→              {saving && <Loader2 className="h-4 w-4 animate-spin me-2" />}
   351→              {form.id ? "حفظ التعديلات" : "إضافة المتجر"}
   352→            </Button>
   353→          </div>
   354→        </DialogContent>
   355→      </Dialog>
   356→
   357→      {/* إدارة المنيو */}
   358→      <MenuManager store={menuStore} onClose={() => setMenuStore(null)} onChanged={load} />
   359→
   360→      {/* تأكيد الحذف */}
   361→      <AlertDialog open={!!deleteTarget} onOpenChange={(o) => !o && setDeleteTarget(null)}>
   362→        <AlertDialogContent dir="rtl">
   363→          <AlertDialogHeader>
   364→            <AlertDialogTitle>حذف «{deleteTarget?.name}»؟</AlertDialogTitle>
   365→            <AlertDialogDescription>
   366→              سيتم حذف المتجر وكل أصنافه نهائياً. لا يمكن التراجع عن هذا الإجراء.
   367→            </AlertDialogDescription>
   368→          </AlertDialogHeader>
   369→          <AlertDialogFooter>
   370→            <AlertDialogCancel>إلغاء</AlertDialogCancel>
   371→            <AlertDialogAction className="bg-red-500 hover:bg-red-600" onClick={remove}>نعم، احذف</AlertDialogAction>
   372→          </AlertDialogFooter>
   373→        </AlertDialogContent>
   374→      </AlertDialog>
   375→    </div>
   376→  );
   377→}

   380→function MenuManager({ store, onClose, onChanged }: { store: Store | null; onClose: () => void; onChanged: () => void }) {
   381→  const [items, setItems] = useState<StoreItem[]>([]);
   382→  const [loading, setLoading] = useState(false);
   383→  const [name, setName] = useState("");
   384→  const [price, setPrice] = useState("");
   385→  const [desc, setDesc] = useState("");
   386→  const [section, setSection] = useState("الأطباق الرئيسية");
   387→  const [imageUrl, setImgUrl] = useState("");
   388→  const [popular, setPopular] = useState(false);
   389→  const [saving, setSaving] = useState(false);
   390→  const [deleteId, setDeleteId] = useState<string | null>(null);
   391→
   392→  const reload = useCallback(() => {
   393→    if (!store) return;
   394→    setLoading(true);
   395→    fetch(`/api/stores/${store.id}`)
   396→      .then((r) => r.json())
   397→      .then((d) => setItems(Array.isArray(d.items) ? d.items : []))
   398→      .finally(() => setLoading(false));
   399→  }, [store]);
   400→
   401→  useEffect(() => {
   402→    reload();
   403→  }, [reload]);
   404→
   405→  const sections = useMemo(() => Array.from(new Set(items.map((i) => i.section))), [items]);
   406→
   407→  async function addItem() {
   408→    if (!store || !name.trim() || !price) {
   409→      toast({ title: "الاسم والسعر مطلوبان", variant: "destructive" });
   410→      return;
   411→    }
   412→    setSaving(true);
   413→    try {
   414→      const res = await fetch("/api/menu-items", {
   415→        method: "POST",
   416→        headers: { "Content-Type": "application/json" },
   417→        body: JSON.stringify({
   418→          storeId: store.id,
   419→          name: name.trim(),
   420→          description: desc.trim() || null,
   421→          price: Number(price),
   422→          section: section.trim() || "الأطباق الرئيسية",
   423→          imageUrl: imageUrl.trim() || null,
   424→          isPopular: popular,
   425→        }),
   426→      });
   427→      if (!res.ok) throw new Error();
   428→      toast({ title: "تمت إضافة الصنف ✅" });
   429→      setName(""); setPrice(""); setDesc(""); setImgUrl(""); setPopular(false);
   430→      reload();
   431→      onChanged();
   432→    } catch {
   433→      toast({ title: "تعذر إضافة الصنف", variant: "destructive" });
   434→    }
   435→    setSaving(false);
   436→  }
   437→
   438→  async function toggleAvailable(it: StoreItem) {
   439→    await fetch(`/api/menu-items/${it.id}`, {
   440→      method: "PATCH",
   441→      headers: { "Content-Type": "application/json" },
   442→      body: JSON.stringify({ isAvailable: !it.isAvailable }),
   443→    });
   444→    reload();
   445→  }
   446→
   447→  async function removeItem() {
   448→    if (!deleteId) return;
   449→    await fetch(`/api/menu-items/${deleteId}`, { method: "DELETE" });
   450→    setDeleteId(null);
   451→    reload();
   452→    onChanged();
   453→  }
   454→
   455→  return (
   456→    <>
   457→      <Dialog open={!!store} onOpenChange={(o) => !o && onClose()}>
   458→        <DialogContent className="max-w-2xl max-h-[88vh] overflow-hidden flex flex-col">
   459→          <DialogHeader>
   460→            <DialogTitle className="font-black flex items-center gap-2">
   461→              <Utensils className="h-5 w-5 text-[#FF6B00]" />
   462→              منيو «{store?.name}»
   463→            </DialogTitle>
   464→            <DialogDescription>أضف الأطباق أو المنتجات وأسعارها — ستظهر فوراً للعملاء</DialogDescription>
   465→          </DialogHeader>
   466→
   467→          {/* نموذج إضافة صنف */}
   468→          <div className="rounded-2xl border bg-[#F8F9FA] p-3.5 space-y-3">
   469→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5">
   470→              <div className="col-span-2 space-y-1">
   471→                <Label htmlFor="mi-name" className="text-xs">اسم الصنف *</Label>
   472→                <Input id="mi-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="مثال: كباب مشكل" className="bg-white h-9" />
   473→              </div>
   474→              <div className="space-y-1">
   475→                <Label htmlFor="mi-price" className="text-xs">السعر (د.ع) *</Label>
   476→                <Input id="mi-price" type="number" value={price} onChange={(e) => setPrice(e.target.value)} placeholder="5000" className="bg-white h-9" />
   477→              </div>
   478→              <div className="space-y-1">
   479→                <Label htmlFor="mi-sec" className="text-xs">القسم</Label>
   480→                <Input id="mi-sec" value={section} onChange={(e) => setSection(e.target.value)} list="sections-list" className="bg-white h-9" />
   481→                <datalist id="sections-list">
   482→                  {["المقبلات", "الأطباق الرئيسية", "المشروبات", "الحلويات", ...sections].map((s) => (
   483→                    <option key={s} value={s} />
   484→                  ))}
   485→                </datalist>
   486→              </div>
   487→            </div>
   488→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-2.5">
   489→              <div className="sm:col-span-2 space-y-1">
   490→                <Label htmlFor="mi-img" className="text-xs">رابط صورة الصنف (اختياري)</Label>
   491→                <Input id="mi-img" value={imageUrl} onChange={(e) => setImgUrl(e.target.value)} dir="ltr" className="bg-white h-9 text-left text-xs" placeholder="https://..." />
   492→              </div>
   493→              <div className="flex items-end pb-0.5">
   494→                <label className="flex items-center gap-2 text-xs font-bold cursor-pointer">
   495→                  <Switch checked={popular} onCheckedChange={setPopular} /> الأكثر طلباً 🔥
   496→                </label>
   497→              </div>
   498→            </div>
   499→            <div className="space-y-1">
   500→              <Label htmlFor="mi-desc" className="text-xs">وصف قصير (اختياري)</Label>
   501→              <Input id="mi-desc" value={desc} onChange={(e) => setDesc(e.target.value)} placeholder="المكونات أو التفاصيل..." className="bg-white h-9" />
   502→            </div>
   503→            <Button className="w-full rounded-full bg-[#FF6B00] hover:bg-[#E85F00] h-9 font-bold" onClick={addItem} disabled={saving}>
   504→              {saving ? <Loader2 className="h-4 w-4 animate-spin me-1.5" /> : <Plus className="h-4 w-4 me-1.5" />}
   505→              إضافة الصنف للمنيو
   506→            </Button>
   507→          </div>
   508→
   509→          {/* قائمة الأصناف */}
   510→          <div className="flex-1 overflow-y-auto nice-scroll mt-3 max-h-[42vh]">
   511→            {loading ? (
   512→              <p className="text-center text-sm text-muted-foreground py-6">جاري التحميل...</p>
   513→            ) : items.length === 0 ? (
   514→              <div className="text-center py-8">
   515→                <PackageOpen className="h-10 w-10 mx-auto text-[#FFAA00] mb-2" />
   516→                <p className="text-sm text-muted-foreground">لا توجد أصناف بعد — أضف أول صنف من الأعلى</p>
   517→              </div>
   518→            ) : (
   519→              <div className="space-y-2 pe-1">
   520→                {items.map((it) => (
   521→                  <div key={it.id} className="flex items-center gap-2.5 bg-white border rounded-xl p-2.5">
   522→                    <div className="w-11 h-11 rounded-lg overflow-hidden bg-[#F8F9FA] shrink-0">
   523→                      <img src={it.imageUrl || "/placeholder-restaurant.svg"} alt={it.name} className="w-full h-full object-cover" />
   524→                    </div>
   525→                    <div className="flex-1 min-w-0">
   526→                      <div className="flex items-center gap-1.5">
   527→                        <span className="font-bold text-sm truncate">{it.name}</span>
   528→                        {it.isPopular && <span className="text-[9px] bg-[#FFF3E6] text-[#FF6B00] px-1 rounded font-bold">🔥</span>}
   529→                      </div>
   530→                      <span className="text-[11px] text-muted-foreground">{it.section}</span>
   531→                    </div>
   532→                    <span className="font-black text-sm text-[#FF6B00] whitespace-nowrap">{formatIQD(it.price)}</span>
   533→                    <Button size="sm" variant="outline" className={`h-7 rounded-full px-2 text-[10px] ${it.isAvailable ? "text-[#2E7D32]" : "text-red-500"}`} onClick={() => toggleAvailable(it)}>
   534→                      {it.isAvailable ? "متوفر" : "نافذ"}
   535→                    </Button>
   536→                    <Button size="sm" variant="outline" className="h-7 w-7 rounded-full p-0 text-red-500" onClick={() => setDeleteId(it.id)} aria-label={`حذف ${it.name}`}>
   537→                      <Trash2 className="h-3 w-3" />
   538→                    </Button>
   539→                  </div>
   540→                ))}
   541→              </div>
   542→            )}
   543→          </div>
   544→        </DialogContent>
   545→      </Dialog>
   546→
   547→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>
   548→        <AlertDialogContent dir="rtl">
   549→          <AlertDialogHeader>
   550→            <AlertDialogTitle>حذف الصنف؟</AlertDialogTitle>
   551→            <AlertDialogDescription>سيُحذف الصنف من المنيو نهائياً.</AlertDialogDescription>
   552→          </AlertDialogHeader>
   553→          <AlertDialogFooter>
   554→            <AlertDialogCancel>إلغاء</AlertDialogCancel>
   555→            <AlertDialogAction className="bg-red-500 hover:bg-red-600" onClick={removeItem}>احذف</AlertDialogAction>
   556→          </AlertDialogFooter>
   557→        </AlertDialogContent>
   558→      </AlertDialog>
   559→    </>
   560→  );
   561→}
