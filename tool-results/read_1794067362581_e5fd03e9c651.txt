     3→// تدفق المصادقة الاحترافي ليجيلك — كما في تطبيقات التوصيل العالمية:
     4→//   1) تسجيل الدخول: رقم الهاتف + كلمة المرور — دخول مباشر بدون رمز تحقق
     5→//      • «نسيت كلمة المرور؟» → استعادتها برمز التحقق عبر رقم الهاتف

     7→//   2) إنشاء حساب: الاسم والهاتف وكلمة المرور، ثم رمز تحقق لتفعيل الرقم
     8→//      (التاجر/الكابتن يكمل نموذج طلب الانضمام بعد إدخال الرمز)
     9→//   3) نسيت كلمة المرور: رمز تحقق عبر الهاتف + كلمة مرور جديدة ثم دخول تلقائي
    10→// مكوّن مشترك: يُعرض في شاشة الترحيب (قبل دخول التطبيق) وفي صفحة الحساب

    40→export type AuthRole = "CUSTOMER" | "MERCHANT" | "DRIVER";
    41→export type AuthMode = "login" | "register" | "forgot";
    42→
    43→interface AuthFlowProps {
    44→  /** الدور المحدد مسبقاً من شاشة الترحيب */
    45→  initialRole?: AuthRole;
    46→  /** إخفاء خطوة «شو نوع حسابك؟» — لأن الاختيار تم في شاشة الترحيب قبلها */
    47→  lockRole?: boolean;
    48→  /** الشاشة الافتراضية عند الفتح — تسجيل الدخول افتراضياً */
    49→  initialMode?: AuthMode;
    50→  /** زر الرجوع — إن لم يُمرَّر لا يظهر زر رجوع */
    51→  onBack?: () => void;
    52→  /** إخفاء بطاقة الهوية الكبيرة (عند العرض داخل شاشة الترحيب التي لها غلافها) */
    53→  showBrandCard?: boolean;
    54→}
    55→
    56→export function AuthFlow({
    57→  initialRole = "CUSTOMER",
    58→  lockRole = false,
    59→  initialMode = "login",
    60→  onBack,
    61→  showBrandCard = true,
    62→}: AuthFlowProps) {
    63→  const setUser = useAuthStore((s) => s.setUser);
    64→  const { toast } = useToast();
    65→
    66→  // الوضع الحالي: دخول / إنشاء حساب / نسيت كلمة المرور
    67→  const [mode, setMode] = useState<AuthMode>(initialMode);
    68→
    69→  // بيانات مشتركة بين الأوضاع — الرقم يبقى محفوظاً عند التنقل بين الشاشات
    70→  const [phone, setPhone] = useState("");
    71→  const [name, setName] = useState("");
    72→  const [password, setPassword] = useState(""); // كلمة مرور إنشاء الحساب
    73→  const [confirm, setConfirm] = useState("");
    74→  const [showPass, setShowPass] = useState(false);
    75→  const [showConfirm, setShowConfirm] = useState(false);
    76→
    77→  // تسجيل الدخول — رقم + كلمة مرور فقط
    78→  const [loginPass, setLoginPass] = useState("");
    79→  const [showLoginPass, setShowLoginPass] = useState(false);
    80→  const [loginHint, setLoginHint] = useState<"NO_PASSWORD" | "NO_USER" | null>(null);
    81→
    82→  // خطوات إنشاء الحساب: البيانات → الرمز → نوع الحساب (لغير المقيّد) → نموذج الانضمام
    83→  const [step, setStep] = useState<"credentials" | "code" | "role" | "apply">("credentials");
    84→  const [code, setCode] = useState("");
    85→  const [codeError, setCodeError] = useState<string | null>(null);
    86→  const [isNewUser, setIsNewUser] = useState(false);
    87→  const [devCode, setDevCode] = useState<string | null>(null);
    88→
    89→  // نسيت كلمة المرور: الرقم → الرمز + كلمة المرور الجديدة
    90→  const [fStep, setFStep] = useState<"phone" | "reset">("phone");
    91→  const [resetCode, setResetCode] = useState("");
    92→  const [newPass, setNewPass] = useState("");
    93→  const [newConfirm, setNewConfirm] = useState("");
    94→  const [showNewPass, setShowNewPass] = useState(false);
    95→  const [showNewConfirm, setShowNewConfirm] = useState(false);
    96→  const [fHint, setFHint] = useState<"NO_USER" | null>(null);
    97→  const [resetError, setResetError] = useState<string | null>(null);
    98→
    99→  const [busy, setBusy] = useState(false);
   100→  const [resendIn, setResendIn] = useState(0);
   101→  // الدور المختار — عميل (فوري) أو تاجر/كابتن (طلب انضمام يراجعه المدير)
   102→  const [chosenRole, setChosenRole] = useState<AuthRole>(initialRole);
   103→
   104→  // عداد إعادة إرسال الرمز (45 ثانية)
   105→  useEffect(() => {
   106→    if (resendIn <= 0) return;
   107→    const t = setTimeout(() => setResendIn((s) => s - 1), 1000);
   108→    return () => clearTimeout(t);
   109→  }, [resendIn]);
   110→
   111→  // ===== التنقل بين الأوضاع مع تصفير حالة كل وضع =====
   112→  function goLogin() {
   113→    setMode("login");
   114→    setLoginHint(null);
   115→    setStep("credentials");
   116→    setCode("");
   117→    setCodeError(null);
   118→    setDevCode(null);
   119→    setFStep("phone");
   120→    setResetCode("");
   121→    setNewPass("");
   122→    setNewConfirm("");
   123→    setResetError(null);
   124→    setFHint(null);
   125→  }
   126→
   127→  function goRegister() {
   128→    setMode("register");
   129→    setLoginHint(null);
   130→    setFHint(null);
   131→    setStep("credentials");
   132→    setCode("");
   133→    setCodeError(null);
   134→    setDevCode(null);
   135→  }
   136→
   137→  function goForgot(autoSend = false) {
   138→    setMode("forgot");
   139→    setLoginHint(null);
   140→    setFStep("phone");
   141→    setFHint(null);
   142→    setResetCode("");
   143→    setNewPass("");
   144→    setNewConfirm("");
   145→    setResetError(null);
   146→    setDevCode(null);
   147→    // عند القدوم من تنبيه «حساب بدون كلمة مرور» — نرسل الرمز تلقائياً للرقم المُدخل
   148→    if (autoSend && phone.trim()) sendCode("RESET_PASSWORD");
   149→  }
   150→
   151→  // ===== إرسال رمز التحقق — للتسجيل أو للاستعادة =====
   152→  async function sendCode(purpose: "REGISTER" | "RESET_PASSWORD") {
   153→    if (!phone.trim()) {
   154→      toast({ title: "أدخل رقم هاتفك أولاً", variant: "destructive" });
   155→      return;
   156→    }
   157→    setBusy(true);
   158→    try {
   159→      const res = await fetch("/api/auth/otp/send", {
   160→        method: "POST",
   161→        headers: { "Content-Type": "application/json" },
   162→        body: JSON.stringify({ phone, purpose }),
   163→      });
   164→      const data = await res.json();
   165→      if (!res.ok) {
   166→        if (purpose === "RESET_PASSWORD" && data.code === "NO_USER") setFHint("NO_USER");
   167→        toast({
   168→          title: "تعذر إرسال الرمز",
   169→          description: data.error,
   170→          variant: "destructive",
   171→        });
   172→        setBusy(false);
   173→        return;
   174→      }
   175→      setFHint(null);
   176→      setIsNewUser(Boolean(data.isNewUser));
   177→      setDevCode(data.devCode || null);
   178→      setResendIn(45);
   179→      setCode("");
   180→      setResetCode("");
   181→      if (purpose === "RESET_PASSWORD") {
   182→        setFStep("reset");
   183→        toast({
   184→          title: "تم إرسال رمز التحقق 📲",
   185→          description: `أدخل الرمز المرسل إلى ${formatPhoneDisplay(data.phone)} لتعيين كلمة مرور جديدة`,
   186→        });
   187→      } else {
   188→        setStep("code");
   189→        toast({
   190→          title: "تم إرسال رمز التحقق 📲",
   191→          description: `أدخل الرمز المرسل إلى ${formatPhoneDisplay(data.phone)}`,
   192→        });
   193→      }
   194→    } catch {
   195→      toast({ title: "خطأ بالشبكة", variant: "destructive" });
   196→    }
   197→    setBusy(false);
   198→  }
   199→
   200→  // ===== إنشاء الحساب / تأكيد الدخول بعد إدخال الرمز =====
   201→  async function verifyRegister(application?: JoinApplication) {
   202→    if (code.length !== 6) {
   203→      toast({ title: "أدخل الرمز المكوّن من 6 أرقام", variant: "destructive" });
   204→      return;
   205→    }
   206→    setBusy(true);
   207→    try {
   208→      const res = await fetch("/api/auth/otp/verify", {
   209→        method: "POST",
   210→        headers: { "Content-Type": "application/json" },
   211→        body: JSON.stringify({ phone, code, name, password, role: chosenRole, application }),
   212→      });
   213→      const data = await res.json();
   214→      if (!res.ok) {
   215→        setCodeError(data.error || "تعذر التحقق من الرمز");
   216→        toast({ title: "تعذر التحقق", description: data.error, variant: "destructive" });
   217→        setBusy(false);
   218→        return;
   219→      }
   220→      setUser(data, data.sessionToken ?? null);
   221→      if (data.joinSubmitted) {
   222→        toast({
   223→          title: `تم استلام طلب انضمامك ${chosenRole === "MERCHANT" ? "كمتجر" : "ككابتن"} 🎉`,
   224→          description: "قيد المراجعة الآن من مدير التطبيق — وتقدر تطلب كعميل بشكل طبيعي حتى الموافقة",
   225→        });
   226→      } else {
   227→        toast({
   228→          title: data.isNewUser ? `تم إنشاء حسابك — أهلاً ${data.name}! 🎉` : `أهلاً بعودتك ${data.name} 👋`,
   229→          description: data.isNewUser ? "حسابك محمي بكلمة مرور — تصفح واطلب بحرية" : "دخلت بنجاح — تصفح واطلب بحرية",
   230→        });
   231→      }
   232→    } catch {
   233→      toast({ title: "خطأ بالشبكة", variant: "destructive" });
   234→    }
   235→    setBusy(false);
   236→  }
   237→
   238→  // زر الخطوة «رمز التحقق» في التسجيل — يعتمد على الدور المحدد مسبقاً
   239→  function codeStepNext() {
   240→    if (!isNewUser && !lockRole) return verifyRegister(); // رقم مسجّل — تأكيد كلمة المرور ودخول
   241→    if (lockRole) {
   242→      // من شاشة الترحيب: العميل يُنشأ فوراً، والتاجر/الكابتن يكملون نموذج الانضمام
   243→      return chosenRole === "CUSTOMER" ? verifyRegister() : setStep("apply");
   244→    }
   245→    return setStep("role"); // حساب جديد — يحدد نوع حسابه
   246→  }
   247→
   248→  const codeNextLabel = lockRole
   249→    ? chosenRole === "CUSTOMER"
   250→      ? isNewUser
   251→        ? "تحقق وأنشئ حسابي"
   252→        : "تحقق وادخل بحسابك"
   253→      : "متابعة لنموذج الانضمام"
   254→    : isNewUser
   255→      ? "متابعة"
   256→      : "تحقق وادخل بحسابك";
   257→
   258→  // ===== تسجيل الدخول — رقم الهاتف وكلمة المرور فقط، بدون رمز =====
   259→  async function doLogin() {
   260→    if (!phone.trim() || !loginPass) {
   261→      toast({ title: "أدخل رقم هاتفك وكلمة المرور", variant: "destructive" });
   262→      return;
   263→    }
   264→    setBusy(true);
   265→    try {
   266→      const res = await fetch("/api/auth/login", {
   267→        method: "POST",
   268→        headers: { "Content-Type": "application/json" },
   269→        body: JSON.stringify({ phone, password: loginPass }),
   270→      });
   271→      const data = await res.json();
   272→      if (!res.ok) {
   273→        setLoginHint(data.code === "NO_PASSWORD" ? "NO_PASSWORD" : data.code === "NO_USER" ? "NO_USER" : null);
   274→        toast({ title: "تعذر تسجيل الدخول", description: data.error, variant: "destructive" });
   275→        setBusy(false);
   276→        return;
   277→      }
   278→      setLoginHint(null);
   279→      setUser(data, data.sessionToken ?? null);
   280→      toast({
   281→        title: `أهلاً بعودتك ${data.name} 👋`,
   282→        description: "دخلت بنجاح — تصفح واطلب بحرية",
   283→      });
   284→    } catch {
   285→      toast({ title: "خطأ بالشبكة", variant: "destructive" });
   286→    }
   287→    setBusy(false);
   288→  }
   289→
   290→  // ===== التحقق من بيانات إنشاء الحساب قبل إرسال الرمز =====
   291→  function registerNext() {
   292→    if (!phone.trim()) {
   293→      toast({ title: "أدخل رقم هاتفك أولاً", variant: "destructive" });
   294→      return;
   295→    }
   296→    if (password.length < 6) {
   297→      toast({ title: "كلمة المرور يجب أن تكون 6 أحرف على الأقل", variant: "destructive" });
   298→      return;
   299→    }
   300→    if (password !== confirm) {
   301→      toast({ title: "كلمتا المرور غير متطابقتين", variant: "destructive" });
   302→      return;
   303→    }
   304→    sendCode("REGISTER");
   305→  }
   306→
   307→  // ===== تعيين كلمة المرور الجديدة بعد التحقق من الرمز =====
   308→  async function submitReset() {
   309→    if (resetCode.length !== 6) {
   310→      toast({ title: "أدخل الرمز المكوّن من 6 أرقام", variant: "destructive" });
   311→      return;
   312→    }
   313→    if (newPass.length < 6) {
   314→      toast({ title: "كلمة المرور يجب أن تكون 6 أحرف على الأقل", variant: "destructive" });
   315→      return;
   316→    }
   317→    if (newPass !== newConfirm) {
   318→      toast({ title: "كلمتا المرور غير متطابقتين", variant: "destructive" });
   319→      return;
   320→    }
   321→    setBusy(true);
   322→    try {
   323→      const res = await fetch("/api/auth/password/reset", {
   324→        method: "POST",
   325→        headers: { "Content-Type": "application/json" },
   326→        body: JSON.stringify({ phone, code: resetCode, password: newPass }),
   327→      });
   328→      const data = await res.json();
   329→      if (!res.ok) {
   330→        setResetError(data.error || "تعذر تعيين كلمة المرور");
   331→        toast({ title: "تعذر تعيين كلمة المرور", description: data.error, variant: "destructive" });
   332→        setBusy(false);
   333→        return;
   334→      }
   335→      setUser(data, data.sessionToken ?? null);
   336→      toast({
   337→        title: "تم تعيين كلمة مرورك 🔐",
   338→        description: `أهلاً ${data.name} — دخلت حسابك مباشرة، من الحين هذا رقمك وكلمتك`,
   339→      });
   340→    } catch {
   341→      toast({ title: "خطأ بالشبكة", variant: "destructive" });
   342→    }
   343→    setBusy(false);
   344→  }
   345→
   346→  return (
   347→    <div className="max-w-md mx-auto px-4 pb-28 pt-4">
   348→      {onBack && (
   349→        <button
   350→          onClick={onBack}
   351→          className="flex items-center gap-1.5 text-sm font-bold text-muted-foreground hover:text-[#0000CE] transition mb-3"
   352→        >
   353→          <ArrowRight className="h-4 w-4" /> رجوع
   354→        </button>
   355→      )}
   356→
   357→      <motion.div initial={{ opacity: 0, y: 14 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.35 }}>
   358→        {showBrandCard && (
   359→          <div className="brand-gradient rounded-3xl p-6 text-center text-white mb-4 overflow-hidden relative shadow-brand">
   360→            <div className="absolute -top-6 -start-6 h-24 w-24 rounded-full bg-white/10" />
   361→            <div className="absolute -bottom-8 -end-4 h-24 w-24 rounded-full bg-white/10" />
   362→            <Image
   363→              src="/brand/yjeelak-icon-v2.png"
   364→              alt="يجيلك Yjeelak"
   365→              width={64}
   366→              height={64}
   367→              className="h-16 w-16 rounded-2xl mx-auto mb-2.5 shadow-lg ring-2 ring-white/40"
   368→            />
   369→            <h1 className="text-xl font-black">أهلاً بك في يجيلك Yjeelak</h1>
   370→            <p className="text-sm text-white/90 mt-1">
   371→              سجّل الدخول برقمك وكلمة مرورك — أو أنشئ حساباً جديداً بدقيقة واحدة 🔐
   372→            </p>
   373→          </div>
   374→        )}
   375→
   376→        <div className="bg-white rounded-2xl border border-border p-4">
   377→          <AnimatePresence mode="wait">
   378→            {/* ===== الوضع 1: تسجيل الدخول — رقم + كلمة مرور ===== */}
   379→            {mode === "login" ? (
   380→              <motion.div
   381→                key="login"
   382→                initial={{ opacity: 0, x: 16 }}
   383→                animate={{ opacity: 1, x: 0 }}
   384→                exit={{ opacity: 0, x: -16 }}
   385→                transition={{ duration: 0.2 }}
   386→                className="space-y-3.5"
   387→              >
   388→                <div>
   389→                  <div className="flex items-center gap-2 text-sm font-black text-[#0B1220]">
   390→                    <LogIn className="h-4.5 w-4.5 text-[#0000CE]" />
   391→                    تسجيل الدخول
   392→                  </div>
   393→                  <p className="text-xs text-muted-foreground mt-1 leading-relaxed">
   394→                    أدخل رقم هاتفك وكلمة المرور — دخول مباشر بدون رمز تحقق
   395→                  </p>
   396→                </div>
   397→
   398→                <PhoneField id="login-phone" value={phone} onChange={setPhone} onEnter={doLogin} />
   399→
   400→                <div className="space-y-1.5">
   401→                  <Label htmlFor="login-pass">كلمة المرور</Label>
   402→                  <PasswordInput
   403→                    id="login-pass"
   404→                    value={loginPass}
   405→                    onChange={setLoginPass}
   406→                    show={showLoginPass}
   407→                    onToggle={() => setShowLoginPass((v) => !v)}
   408→                    placeholder="كلمة مرور حسابك"
   409→                    autoComplete="current-password"
   410→                    onEnter={doLogin}
   411→                  />
   412→                  <div className="flex justify-end">
   413→                    <button
   414→                      onClick={() => goForgot()}
   415→                      className="text-xs font-black text-[#0000CE] hover:underline flex items-center gap-1"
   416→                    >
   417→                      <KeyRound className="h-3.5 w-3.5" />
   418→                      نسيت كلمة المرور؟
   419→                    </button>
   420→                  </div>
   421→                </div>
   422→
   423→                {/* تنبيهات توجيهية — حساب قديم بدون كلمة مرور، أو رقم غير مسجل */}
   424→                {loginHint === "NO_PASSWORD" && (
   425→                  <div className="rounded-xl bg-[#FFF7E0] border border-[#F5A623]/50 p-3 space-y-2">
   426→                    <p className="text-[11.5px] font-semibold text-[#0B1220] leading-relaxed">
   427→                      🔐 حسابك سُجِّل سابقاً برمز التحقق وبدون كلمة مرور — اضبطها الآن برمز التحقق
   428→                      المرسل إلى رقمك، ومن بعدها تدخل مباشرة بكلمة المرور
   429→                    </p>
   430→                    <Button
   431→                      size="sm"
   432→                      className="w-full h-9 rounded-full bg-[#0000CE] hover:bg-[#0000A3] font-black text-xs"
   433→                      onClick={() => goForgot(true)}
   434→                      disabled={busy}
   435→                    >
   436→                      {busy ? <Loader2 className="h-4 w-4 animate-spin me-1" /> : <KeyRound className="h-4 w-4 me-1" />}
   437→                      ضبط كلمة المرور عبر رقم الهاتف
   438→                    </Button>
   439→                  </div>
   440→                )}
