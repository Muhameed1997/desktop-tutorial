     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useState } from "react";
     4→import { Plus, Pencil, Trash2, Utensils, PackageOpen, Power, Search, Loader2, Star, MapPin } from "lucide-react";
     5→import { Button } from "@/components/ui/button";
     6→import { Input } from "@/components/ui/input";
     7→import { Label } from "@/components/ui/label";
     8→import { Textarea } from "@/components/ui/textarea";
     9→import { Switch } from "@/components/ui/switch";
    10→import { Badge } from "@/components/ui/badge";
    11→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription } from "@/components/ui/dialog";
    12→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    13→import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle } from "@/components/ui/alert-dialog";
    14→import { Skeleton } from "@/components/ui/skeleton";
    15→import { toast } from "@/hooks/use-toast";
    16→import { formatIQD } from "@/lib/format";
    17→import { TYPE_AR, type Store, type StoreItem, type StoreType } from "@/lib/types";
    18→import { HADITHA_CENTER } from "@/lib/geo";
    19→import LocationPickerMap from "./location-picker-map";
    20→import { adminPinHeaders } from "./admin-join";
    21→
    22→const TYPE_OPTIONS: { value: StoreType; label: string; icon: string }[] = [
    23→  { value: "RESTAURANT", label: "مطعم", icon: "🍽️" },
    24→  { value: "STORE", label: "متجر", icon: "🏬" },
    25→  { value: "MARKET", label: "سوق", icon: "🛒" },
    26→];
    27→
    28→const CATEGORY_PRESETS: Record<StoreType, string[]> = {
    29→  RESTAURANT: ["عراقي", "برجر", "بيتزا", "شاورما", "دجاج ومشويات", "أسماك", "حلويات", "قهوة ومشروبات", "فطور صباحي"],
    30→  STORE: ["حلويات ومخبوزات", "عصائر ومثلجات", "بقالة", "إلكترونيات", "ملابس", "صيدلية"],
    31→  MARKET: ["خضار وفواكه", "سوبرماركت", "تمر ومكسرات", "لحوم ودواجن", "منتجات ألبان"],
    32→};
    33→
    34→interface StoreForm {
    35→  id?: string;
    36→  name: string;
    37→  description: string;
    38→  type: StoreType;
    39→  category: string;
    40→  imageUrl: string;
    41→  deliveryFee: string;
    42→  deliveryTime: string;
    43→  minOrder: string;
    44→  phone: string;
    45→  address: string;
    46→  lat: string;
    47→  lng: string;
    48→  isOpen: boolean;
    49→  isFeatured: boolean;
    50→}
    51→
    52→const EMPTY_FORM: StoreForm = {
    53→  name: "",
    54→  description: "",
    55→  type: "RESTAURANT",
    56→  category: "عراقي",
    57→  imageUrl: "",
    58→  deliveryFee: "1000",
    59→  deliveryTime: "30-45 دقيقة",
    60→  minOrder: "0",
    61→  phone: "",
    62→  address: "",
    63→  lat: String(HADITHA_CENTER.lat),
    64→  lng: String(HADITHA_CENTER.lng),
    65→  isOpen: true,
    66→  isFeatured: false,
    67→};
    68→
    69→export function AdminStoresTab() {
    70→  const [stores, setStores] = useState<Store[]>([]);
    71→  const [loading, setLoading] = useState(true);
    72→  const [dialogOpen, setDialogOpen] = useState(false);
    73→  const [form, setForm] = useState<StoreForm>(EMPTY_FORM);
    74→  const [saving, setSaving] = useState(false);
    75→  const [deleteTarget, setDeleteTarget] = useState<Store | null>(null);
    76→  const [menuStore, setMenuStore] = useState<Store | null>(null);
    77→  const [filter, setFilter] = useState<"ALL" | StoreType>("ALL");
    78→  const [search, setSearch] = useState("");
    79→
    80→  const load = useCallback(() => {
    81→    setLoading(true);
    82→    fetch("/api/stores")
    83→      .then((r) => r.json())
    84→      .then((d) => setStores(Array.isArray(d) ? d : []))
    85→      .catch(() => toast({ title: "خطأ في جلب المتاجر", variant: "destructive" }))
    86→      .finally(() => setLoading(false));
    87→  }, []);
    88→
    89→  useEffect(() => {
    90→    load();
    91→  }, [load]);
    92→
    93→  const filtered = useMemo(() => {
    94→    return stores.filter(
    95→      (s) =>
    96→        (filter === "ALL" || s.type === filter) &&
    97→        (!search.trim() || s.name.includes(search.trim()) || s.category.includes(search.trim()))
    98→    );
    99→  }, [stores, filter, search]);
   100→
   101→  function openCreate() {
   102→    setForm(EMPTY_FORM);
   103→    setDialogOpen(true);
   104→  }
   105→
   106→  function openEdit(s: Store) {
   107→    setForm({
   108→      id: s.id,
   109→      name: s.name,
   110→      description: s.description || "",
   111→      type: s.type,
   112→      category: s.category,
   113→      imageUrl: s.imageUrl || "",
   114→      deliveryFee: String(s.deliveryFee),
   115→      deliveryTime: s.deliveryTime,
   116→      minOrder: String(s.minOrder),
   117→      phone: s.phone || "",
   118→      address: s.address || "",
   119→      lat: s.lat != null ? String(s.lat) : String(HADITHA_CENTER.lat),
   120→      lng: s.lng != null ? String(s.lng) : String(HADITHA_CENTER.lng),
   121→      isOpen: s.isOpen,
   122→      isFeatured: s.isFeatured,
   123→    });
   124→    setDialogOpen(true);
   125→  }
   126→
   127→  async function save() {
   128→    if (!form.name.trim()) {
   129→      toast({ title: "اسم المتجر مطلوب", variant: "destructive" });
   130→      return;
   131→    }
   132→    setSaving(true);
   133→    const payload = {
   134→      name: form.name.trim(),
   135→      description: form.description.trim() || null,
   136→      type: form.type,
   137→      category: form.category,
   138→      imageUrl: form.imageUrl.trim() || null,
   139→      deliveryFee: Number(form.deliveryFee) || 0,
   140→      deliveryTime: form.deliveryTime.trim() || "30-45 دقيقة",
   141→      minOrder: Number(form.minOrder) || 0,
   142→      phone: form.phone.trim() || null,
   143→      address: form.address.trim() || null,
   144→      lat: Number(form.lat) || 34.1329,
   145→      lng: Number(form.lng) || 42.3775,
   146→      isOpen: form.isOpen,
   147→      isFeatured: form.isFeatured,
   148→    };
   149→    try {
   150→      const res = await fetch(form.id ? `/api/stores/${form.id}` : "/api/stores", {
   151→        method: form.id ? "PATCH" : "POST",
   152→        headers: { "Content-Type": "application/json", ...adminPinHeaders() },
   153→        body: JSON.stringify(payload),
   154→      });
   155→      const data = await res.json();
   156→      if (!res.ok) throw new Error(data.error);
   157→      toast({ title: form.id ? "تم تعديل المتجر ✅" : "تمت إضافة المتجر ✅" });
   158→      setDialogOpen(false);
   159→      load();
   160→    } catch (e) {
   161→      toast({ title: "خطأ في الحفظ", description: e instanceof Error ? e.message : "حاول مجدداً", variant: "destructive" });
   162→    }
   163→    setSaving(false);
   164→  }
   165→
   166→  async function toggleOpen(s: Store) {
   167→    await fetch(`/api/stores/${s.id}`, {
   168→      method: "PATCH",
   169→      headers: { "Content-Type": "application/json", ...adminPinHeaders() },
   170→      body: JSON.stringify({ isOpen: !s.isOpen }),
   171→    });
   172→    toast({ title: !s.isOpen ? `${s.name}: مفتوح الآن 🟢` : `${s.name}: مغلق الآن 🔴` });
   173→    load();
   174→  }
   175→
   176→  async function remove() {
   177→    if (!deleteTarget) return;
   178→    await fetch(`/api/stores/${deleteTarget.id}`, { method: "DELETE", headers: adminPinHeaders() });
   179→    toast({ title: `تم حذف "${deleteTarget.name}"` });
   180→    setDeleteTarget(null);
   181→    load();
   182→  }
   183→
   184→  return (
   185→    <div className="space-y-4">
   186→      {/* شريط الأدوات */}
   187→      <div className="flex flex-wrap items-center gap-2">
   188→        <Select value={filter} onValueChange={(v) => setFilter(v as "ALL" | StoreType)}>
   189→          <SelectTrigger className="w-36 rounded-full bg-white h-10">
   190→            <SelectValue />
   191→          </SelectTrigger>
   192→          <SelectContent>
   193→            <SelectItem value="ALL">كل الأنواع</SelectItem>
   194→            {TYPE_OPTIONS.map((t) => (
   195→              <SelectItem key={t.value} value={t.value}>{t.icon} {t.label}</SelectItem>
   196→            ))}
   197→          </SelectContent>
   198→        </Select>
   199→        <div className="relative flex-1 min-w-40">
   200→          <Search className="absolute start-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   201→          <Input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="ابحث بالاسم أو التصنيف..." className="ps-9 rounded-full bg-white h-10" />
   202→        </div>
   203→        <Button className="rounded-full bg-[#0000CE] hover:bg-[#0000A3] h-10 px-4 font-bold" onClick={openCreate}>
   204→          <Plus className="h-4 w-4 me-1.5" /> إضافة متجر
   205→        </Button>
   206→      </div>
   207→
   208→      {/* القائمة */}
   209→      {loading ? (
   210→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-2xl" />)}</div>
   211→      ) : filtered.length === 0 ? (
   212→        <div className="bg-white rounded-2xl p-10 text-center border border-dashed border-[#00C46A]">
   213→          <div className="text-4xl mb-2">🏪</div>
   214→          <p className="font-bold">لا توجد متاجر مطابقة</p>
   215→          <p className="text-sm text-muted-foreground mt-1">اضغط «إضافة متجر» لإنشاء أول مطعم أو متجر أو سوق</p>
   216→        </div>
   217→      ) : (
   218→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   219→          {filtered.map((s) => (
   220→            <div key={s.id} className="bg-white rounded-2xl border border-border p-3.5 flex gap-3">
   221→              <div className="w-16 h-16 rounded-xl overflow-hidden shrink-0 bg-[#EAEDFF]">
   222→                <img src={s.imageUrl || (s.type === "MARKET" ? "/placeholder-market.svg" : s.type === "STORE" ? "/placeholder-store.svg" : "/placeholder-restaurant.svg")} alt={s.name} className="w-full h-full object-cover" />
   223→              </div>
   224→              <div className="flex-1 min-w-0">
   225→                {/* min-w-0 ضرورية: بدونها لا يتقلص الاسم ويخرج شارا «مطعم/رواج» خارج البطاقة في الشاشات الضيقة */}
   226→                <div className="flex items-center gap-1.5 flex-wrap min-w-0">
   227→                  <h4 className="font-black text-sm truncate min-w-0">{s.name}</h4>
   228→                  <Badge className={`border-0 text-[10px] shrink-0 ${s.type === "MARKET" ? "bg-[#DFFFEF] text-[#00875A]" : s.type === "STORE" ? "bg-[#EEF1F6] text-[#3A4356]" : "bg-[#EAEDFF] text-[#0000A3]"}`}>{TYPE_AR[s.type]}</Badge>
   229→                  {s.isFeatured && <Badge className="border-0 text-[10px] bg-[#CFFFE2] text-[#3A4356] shrink-0"><Star className="h-2.5 w-2.5 me-0.5 fill-current" /> رواج</Badge>}
   230→                </div>
   231→                <p className="text-[11px] text-muted-foreground mt-0.5 leading-snug">{s.category} · {s.deliveryTime} · توصيل {formatIQD(s.deliveryFee)} · {s.items?.length || 0} صنف</p>
   232→                <div className="flex items-center gap-1.5 mt-2">
   233→                  <Button size="sm" variant="outline" className="h-7 rounded-full text-[11px] px-2.5" onClick={() => openEdit(s)}>
   234→                    <Pencil className="h-3 w-3 me-1" /> تعديل
   235→                  </Button>
   236→                  <Button size="sm" variant="outline" className="h-7 rounded-full text-[11px] px-2.5" onClick={() => setMenuStore(s)}>
   237→                    <Utensils className="h-3 w-3 me-1" /> المنيو ({s.items?.length || 0})
   238→                  </Button>
   239→                  <Button
   240→                    size="sm"
   241→                    variant="outline"
   242→                    className={`h-7 rounded-full text-[11px] px-2.5 ${s.isOpen ? "text-[#00875A]" : "text-red-500"}`}
   243→                    onClick={() => toggleOpen(s)}
   244→                  >
   245→                    <Power className="h-3 w-3 me-1" /> {s.isOpen ? "مفتوح" : "مغلق"}
   246→                  </Button>
   247→                  <Button size="sm" variant="outline" className="h-7 w-7 rounded-full p-0 text-red-500 hover:bg-red-50" onClick={() => setDeleteTarget(s)} aria-label={`حذف ${s.name}`}>
   248→                    <Trash2 className="h-3.5 w-3.5" />
   249→                  </Button>
   250→                </div>
   251→              </div>
   252→            </div>
   253→          ))}
   254→        </div>
   255→      )}
   256→
   257→      {/* نافذة الإضافة/التعديل */}
   258→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   259→        <DialogContent className="max-w-lg max-h-[88vh] overflow-y-auto nice-scroll">
   260→          <DialogHeader>
   261→            <DialogTitle className="font-black">{form.id ? `تعديل: ${form.name}` : "إضافة متجر جديد"}</DialogTitle>
   262→            <DialogDescription>أضف مطعماً أو متجراً أو سوقاً وسيظهر مباشرة في واجهة العملاء</DialogDescription>
   263→          </DialogHeader>
   264→          <div className="space-y-3.5">
   265→            <div className="space-y-1.5">
   266→              <Label>نوع المتجر *</Label>
   267→              <div className="grid grid-cols-3 gap-2">
   268→                {TYPE_OPTIONS.map((t) => (
   269→                  <button
   270→                    key={t.value}
   271→                    type="button"
   272→                    onClick={() => { setForm((f) => ({ ...f, type: t.value, category: CATEGORY_PRESETS[t.value][0] })); }}
   273→                    className={`rounded-xl border p-2.5 text-center text-xs font-bold transition ${form.type === t.value ? "border-[#0000CE] bg-[#EAEDFF]" : "border-border hover:border-[#00C46A]"}`}
   274→                  >
   275→                    <div className="text-lg">{t.icon}</div>
   276→                    {t.label}
   277→                  </button>
   278→                ))}
   279→              </div>
   280→            </div>
   281→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   282→              <div className="space-y-1.5">
   283→                <Label htmlFor="st-name">الاسم *</Label>
   284→                <Input id="st-name" value={form.name} onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))} placeholder="مثال: بيت الكبابي" className="bg-[#EAEDFF]" />
   285→              </div>
   286→              <div className="space-y-1.5">
   287→                <Label>التصنيف</Label>
   288→                <Select value={form.category} onValueChange={(v) => setForm((f) => ({ ...f, category: v }))}>
   289→                  <SelectTrigger className="bg-[#EAEDFF]"><SelectValue /></SelectTrigger>
   290→                  <SelectContent>
   291→                    {CATEGORY_PRESETS[form.type].map((c) => (
   292→                      <SelectItem key={c} value={c}>{c}</SelectItem>
   293→                    ))}
   294→                  </SelectContent>
   295→                </Select>
   296→              </div>
   297→            </div>
   298→            <div className="space-y-1.5">
   299→              <Label htmlFor="st-desc">الوصف</Label>
   300→              <Textarea id="st-desc" value={form.description} onChange={(e) => setForm((f) => ({ ...f, description: e.target.value }))} placeholder="وصف قصير يظهر للعملاء..." className="bg-[#EAEDFF] min-h-16" />
   301→            </div>
   302→            <div className="space-y-1.5">
   303→              <Label htmlFor="st-img">رابط صورة المتجر</Label>
   304→              <Input id="st-img" value={form.imageUrl} onChange={(e) => setForm((f) => ({ ...f, imageUrl: e.target.value }))} placeholder="https://... (اتركه فارغاً لاستخدام صورة افتراضية)" dir="ltr" className="bg-[#EAEDFF] text-left text-xs" />
   305→              {form.imageUrl.trim() && (
   306→                <div className="h-24 w-40 rounded-xl overflow-hidden border">
   307→                  <img src={form.imageUrl} alt="معاينة" className="w-full h-full object-cover" />
   308→                </div>
   309→              )}
   310→            </div>
   311→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-3">
   312→              <div className="space-y-1.5">
   313→                <Label htmlFor="st-fee">أجرة التوصيل (د.ع)</Label>
   314→                <Input id="st-fee" type="number" value={form.deliveryFee} onChange={(e) => setForm((f) => ({ ...f, deliveryFee: e.target.value }))} className="bg-[#EAEDFF]" />
   315→              </div>
   316→              <div className="space-y-1.5">
   317→                <Label htmlFor="st-min">أدنى طلب (د.ع)</Label>
   318→                <Input id="st-min" type="number" value={form.minOrder} onChange={(e) => setForm((f) => ({ ...f, minOrder: e.target.value }))} className="bg-[#EAEDFF]" />
   319→              </div>
   320→              <div className="space-y-1.5">
   321→                <Label htmlFor="st-time">وقت التوصيل</Label>
   322→                <Input id="st-time" value={form.deliveryTime} onChange={(e) => setForm((f) => ({ ...f, deliveryTime: e.target.value }))} className="bg-[#EAEDFF]" />
   323→              </div>
   324→            </div>
   325→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   326→              <div className="space-y-1.5">
   327→                <Label htmlFor="st-phone">هاتف المتجر</Label>
   328→                <Input id="st-phone" value={form.phone} onChange={(e) => setForm((f) => ({ ...f, phone: e.target.value }))} placeholder="07XXXXXXXXX" dir="ltr" className="bg-[#EAEDFF] text-left" />
   329→              </div>
   330→              <div className="space-y-1.5">
   331→                <Label htmlFor="st-addr">العنوان</Label>
   332→                <Input id="st-addr" value={form.address} onChange={(e) => setForm((f) => ({ ...f, address: e.target.value }))} placeholder="حديثة – حي المعلمين" className="bg-[#EAEDFF]" />
   333→              </div>
   334→            </div>
   335→            <div className="space-y-1.5">
   336→              <Label className="flex items-center gap-1.5">
   337→                <MapPin className="h-3.5 w-3.5 text-[#0000CE]" /> موقع المتجر على الخريطة
   338→              </Label>
   339→              <div className="rounded-xl overflow-hidden border border-border">
   340→                <LocationPickerMap
   341→                  value={{ lat: Number(form.lat) || HADITHA_CENTER.lat, lng: Number(form.lng) || HADITHA_CENTER.lng }}
   342→                  onChange={(p) => setForm((f) => ({ ...f, lat: p.lat.toFixed(6), lng: p.lng.toFixed(6) }))}
   343→                  className="h-56 w-full"
   344→                />
   345→              </div>
   346→              <p className="text-[11px] text-muted-foreground flex items-center gap-1">
   347→                <MapPin className="h-3 w-3 text-[#0000CE]" />
   348→                انقر على الخريطة أو اسحب الدبوس 📍 — يُحفظ الموقع تلقائياً داخل مدينة حديثة
   349→                <span dir="ltr" className="font-bold">({form.lat}, {form.lng})</span>
   350→              </p>
   351→            </div>
   352→            <div className="flex items-center justify-between rounded-xl border p-3">
   353→              <Label htmlFor="st-open" className="cursor-pointer">المتجر مفتوح لاستقبال الطلبات</Label>
   354→              <Switch id="st-open" checked={form.isOpen} onCheckedChange={(v) => setForm((f) => ({ ...f, isOpen: v }))} />
   355→            </div>
   356→            <div className="flex items-center justify-between rounded-xl border p-3">
   357→              <Label htmlFor="st-feat" className="cursor-pointer">عرضه ضمن «الأكثر رواجاً»</Label>
   358→              <Switch id="st-feat" checked={form.isFeatured} onCheckedChange={(v) => setForm((f) => ({ ...f, isFeatured: v }))} />
   359→            </div>
   360→            <Button className="w-full rounded-full bg-[#0000CE] hover:bg-[#0000A3] h-11 font-black" onClick={save} disabled={saving}>
   361→              {saving && <Loader2 className="h-4 w-4 animate-spin me-2" />}
   362→              {form.id ? "حفظ التعديلات" : "إضافة المتجر"}
   363→            </Button>
   364→          </div>
   365→        </DialogContent>
   366→      </Dialog>
   367→
   368→      {/* إدارة المنيو */}
   369→      <MenuManager store={menuStore} onClose={() => setMenuStore(null)} onChanged={load} />
   370→
   371→      {/* تأكيد الحذف */}
   372→      <AlertDialog open={!!deleteTarget} onOpenChange={(o) => !o && setDeleteTarget(null)}>
   373→        <AlertDialogContent dir="rtl">
   374→          <AlertDialogHeader>
   375→            <AlertDialogTitle>حذف «{deleteTarget?.name}»؟</AlertDialogTitle>
   376→            <AlertDialogDescription>
   377→              سيتم حذف المتجر وكل أصنافه نهائياً. لا يمكن التراجع عن هذا الإجراء.
   378→            </AlertDialogDescription>
   379→          </AlertDialogHeader>
   380→          <AlertDialogFooter>
   381→            <AlertDialogCancel>إلغاء</AlertDialogCancel>
   382→            <AlertDialogAction className="bg-red-500 hover:bg-red-600" onClick={remove}>نعم، احذف</AlertDialogAction>
   383→          </AlertDialogFooter>
   384→        </AlertDialogContent>
   385→      </AlertDialog>
   386→    </div>
   387→  );
   388→}

   390→// ===== إدارة أصناف المنيو لمتجر معين =====
   391→function MenuManager({ store, onClose, onChanged }: { store: Store | null; onClose: () => void; onChanged: () => void }) {
   392→  const [items, setItems] = useState<StoreItem[]>([]);
   393→  const [loading, setLoading] = useState(false);
   394→  const [name, setName] = useState("");
   395→  const [price, setPrice] = useState("");
   396→  const [desc, setDesc] = useState("");
   397→  const [section, setSection] = useState("الأطباق الرئيسية");
   398→  const [imageUrl, setImgUrl] = useState("");
   399→  const [popular, setPopular] = useState(false);
   400→  const [saving, setSaving] = useState(false);
   401→  const [deleteId, setDeleteId] = useState<string | null>(null);
   402→
   403→  const reload = useCallback(() => {
   404→    if (!store) return;
   405→    setLoading(true);
   406→    fetch(`/api/stores/${store.id}`)
   407→      .then((r) => r.json())
   408→      .then((d) => setItems(Array.isArray(d.items) ? d.items : []))
   409→      .finally(() => setLoading(false));
   410→  }, [store]);
   411→
   412→  useEffect(() => {
   413→    reload();
   414→  }, [reload]);
   415→
   416→  const sections = useMemo(() => Array.from(new Set(items.map((i) => i.section))), [items]);
   417→
   418→  async function addItem() {
   419→    if (!store || !name.trim() || !price) {
   420→      toast({ title: "الاسم والسعر مطلوبان", variant: "destructive" });
   421→      return;
   422→    }
   423→    setSaving(true);
   424→    try {
   425→      const res = await fetch("/api/menu-items", {
   426→        method: "POST",
   427→        headers: { "Content-Type": "application/json", ...adminPinHeaders() },
   428→        body: JSON.stringify({
   429→          storeId: store.id,
   430→          name: name.trim(),
   431→          description: desc.trim() || null,
   432→          price: Number(price),
   433→          section: section.trim() || "الأطباق الرئيسية",
   434→          imageUrl: imageUrl.trim() || null,
   435→          isPopular: popular,
   436→        }),
   437→      });
   438→      if (!res.ok) throw new Error();
   439→      toast({ title: "تمت إضافة الصنف ✅" });
   440→      setName(""); setPrice(""); setDesc(""); setImgUrl(""); setPopular(false);
   441→      reload();
   442→      onChanged();
   443→    } catch {
   444→      toast({ title: "تعذر إضافة الصنف", variant: "destructive" });
   445→    }
   446→    setSaving(false);
   447→  }
   448→
   449→  async function toggleAvailable(it: StoreItem) {
   450→    await fetch(`/api/menu-items/${it.id}`, {
   451→      method: "PATCH",
   452→      headers: { "Content-Type": "application/json", ...adminPinHeaders() },
   453→      body: JSON.stringify({ isAvailable: !it.isAvailable }),
   454→    });
   455→    reload();
   456→  }
   457→
   458→  async function removeItem() {
   459→    if (!deleteId) return;
   460→    await fetch(`/api/menu-items/${deleteId}`, { method: "DELETE", headers: adminPinHeaders() });
   461→    setDeleteId(null);
   462→    reload();
   463→    onChanged();
   464→  }
   465→
   466→  return (
   467→    <>
   468→      <Dialog open={!!store} onOpenChange={(o) => !o && onClose()}>
   469→        <DialogContent className="max-w-2xl max-h-[88vh] overflow-hidden flex flex-col">
   470→          <DialogHeader>
   471→            <DialogTitle className="font-black flex items-center gap-2">
   472→              <Utensils className="h-5 w-5 text-[#0000CE]" />
   473→              منيو «{store?.name}»
   474→            </DialogTitle>
   475→            <DialogDescription>أضف الأطباق أو المنتجات وأسعارها — ستظهر فوراً للعملاء</DialogDescription>
   476→          </DialogHeader>
   477→
   478→          {/* نموذج إضافة صنف */}
   479→          <div className="rounded-2xl border bg-[#EAEDFF] p-3.5 space-y-3">
   480→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5">
   481→              <div className="col-span-2 space-y-1">
   482→                <Label htmlFor="mi-name" className="text-xs">اسم الصنف *</Label>
   483→                <Input id="mi-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="مثال: كباب مشكل" className="bg-white h-9" />
   484→              </div>
   485→              <div className="space-y-1">
   486→                <Label htmlFor="mi-price" className="text-xs">السعر (د.ع) *</Label>
   487→                <Input id="mi-price" type="number" value={price} onChange={(e) => setPrice(e.target.value)} placeholder="5000" className="bg-white h-9" />
   488→              </div>
   489→              <div className="space-y-1">
   490→                <Label htmlFor="mi-sec" className="text-xs">القسم</Label>
   491→                <Input id="mi-sec" value={section} onChange={(e) => setSection(e.target.value)} list="sections-list" className="bg-white h-9" />
   492→                <datalist id="sections-list">
   493→                  {/* Array.from(new Set) — لمنع تكرار المفتاح عندما يوجد قسم بنفس الاسم في القوائم الافتراضية والأصناف الحالية */}
   494→                  {Array.from(new Set(["المقبلات", "الأطباق الرئيسية", "المشروبات", "الحلويات", ...sections])).map((s) => (
   495→                    <option key={s} value={s} />
   496→                  ))}
   497→                </datalist>
   498→              </div>
   499→            </div>
   500→            <div className="grid grid-cols-1 sm:grid-cols-3 gap-2.5">
   501→              <div className="sm:col-span-2 space-y-1">
   502→                <Label htmlFor="mi-img" className="text-xs">رابط صورة الصنف (اختياري)</Label>
   503→                <Input id="mi-img" value={imageUrl} onChange={(e) => setImgUrl(e.target.value)} dir="ltr" className="bg-white h-9 text-left text-xs" placeholder="https://..." />
   504→              </div>
   505→              <div className="flex items-end pb-0.5">
   506→                <label className="flex items-center gap-2 text-xs font-bold cursor-pointer">
   507→                  <Switch checked={popular} onCheckedChange={setPopular} /> الأكثر طلباً 🔥
   508→                </label>
   509→              </div>
   510→            </div>
   511→            <div className="space-y-1">
   512→              <Label htmlFor="mi-desc" className="text-xs">وصف قصير (اختياري)</Label>
   513→              <Input id="mi-desc" value={desc} onChange={(e) => setDesc(e.target.value)} placeholder="المكونات أو التفاصيل..." className="bg-white h-9" />
   514→            </div>
   515→            <Button className="w-full rounded-full bg-[#0000CE] hover:bg-[#0000A3] h-9 font-bold" onClick={addItem} disabled={saving}>
   516→              {saving ? <Loader2 className="h-4 w-4 animate-spin me-1.5" /> : <Plus className="h-4 w-4 me-1.5" />}
   517→              إضافة الصنف للمنيو
   518→            </Button>
   519→          </div>
   520→
   521→          {/* قائمة الأصناف */}
   522→          <div className="flex-1 overflow-y-auto nice-scroll mt-3 max-h-[42vh]">
   523→            {loading ? (
   524→              <p className="text-center text-sm text-muted-foreground py-6">جاري التحميل...</p>
   525→            ) : items.length === 0 ? (
   526→              <div className="text-center py-8">
   527→                <PackageOpen className="h-10 w-10 mx-auto text-[#00C46A] mb-2" />
   528→                <p className="text-sm text-muted-foreground">لا توجد أصناف بعد — أضف أول صنف من الأعلى</p>
   529→              </div>
   530→            ) : (
   531→              <div className="space-y-2 pe-1">
   532→                {items.map((it) => (
   533→                  <div key={it.id} className="flex items-center gap-2.5 bg-white border rounded-xl p-2.5">
   534→                    <div className="w-11 h-11 rounded-lg overflow-hidden bg-[#EAEDFF] shrink-0">
   535→                      <img src={it.imageUrl || "/placeholder-restaurant.svg"} alt={it.name} className="w-full h-full object-cover" />
   536→                    </div>
   537→                    <div className="flex-1 min-w-0">
   538→                      <div className="flex items-center gap-1.5">
   539→                        <span className="font-bold text-sm truncate">{it.name}</span>
   540→                        {it.isPopular && <span className="text-[9px] bg-[#EAEDFF] text-[#0000CE] px-1 rounded font-bold">🔥</span>}
   541→                      </div>
   542→                      <span className="text-[11px] text-muted-foreground">{it.section}</span>
   543→                    </div>
   544→                    <span className="font-black text-sm text-[#0000CE] whitespace-nowrap">{formatIQD(it.price)}</span>
   545→                    <Button size="sm" variant="outline" className={`h-7 rounded-full px-2 text-[10px] ${it.isAvailable ? "text-[#00875A]" : "text-red-500"}`} onClick={() => toggleAvailable(it)}>
   546→                      {it.isAvailable ? "متوفر" : "نافذ"}
   547→                    </Button>
   548→                    <Button size="sm" variant="outline" className="h-7 w-7 rounded-full p-0 text-red-500" onClick={() => setDeleteId(it.id)} aria-label={`حذف ${it.name}`}>
   549→                      <Trash2 className="h-3 w-3" />
   550→                    </Button>
   551→                  </div>
   552→                ))}
   553→              </div>
   554→            )}
   555→          </div>
   556→        </DialogContent>
   557→      </Dialog>
   558→
   559→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>
   560→        <AlertDialogContent dir="rtl">
   561→          <AlertDialogHeader>
   562→            <AlertDialogTitle>حذف الصنف؟</AlertDialogTitle>
   563→            <AlertDialogDescription>سيُحذف الصنف من المنيو نهائياً.</AlertDialogDescription>
   564→          </AlertDialogHeader>
   565→          <AlertDialogFooter>
   566→            <AlertDialogCancel>إلغاء</AlertDialogCancel>
   567→            <AlertDialogAction className="bg-red-500 hover:bg-red-600" onClick={removeItem}>احذف</AlertDialogAction>
   568→          </AlertDialogFooter>
   569→        </AlertDialogContent>
   570→      </AlertDialog>
   571→    </>
   572→  );
   573→}
