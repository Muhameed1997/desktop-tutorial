   230→   230→        });

   233→   233→      toast({ title: "خطأ بالشبكة", variant: "destructive" });

   339→   339→      });
   340→   340→    } catch {
   341→   341→      toast({ title: "خطأ بالشبكة", variant: "destructive" });
   342→   342→    }
   343→   343→    setBusy(false);
   344→   344→  }

   362→   362→            <Image

   365→   365→              width={64}

   367→   367→              className="h-16 w-16 rounded-2xl mx-auto mb-2.5 shadow-lg ring-2 ring-white/40"
   368→   368→            />
   369→   369→            <h1 className="text-xl font-black">أهلاً بك في يجيلك Yjeelak</h1>
   370→   370→            <p className="text-sm text-white/90 mt-1">
   371→   371→              سجّل الدخول برقمك وكلمة مرورك — أو أنشئ حساباً جديداً بدقيقة واحدة 🔐
   372→   372→            </p>
   373→   373→          </div>
   374→   374→        )}
   375→   375→
   376→   376→        <div className="bg-white rounded-2xl border border-border p-4">
   377→   377→          <AnimatePresence mode="wait">
   378→   378→            {/* ===== الوضع 1: تسجيل الدخول — رقم + كلمة مرور ===== */}
   379→   379→            {mode === "login" ? (
