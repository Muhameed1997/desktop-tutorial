   424→function ProfileCard() {
   425→  const user = useAuthStore((s) => s.user)!;
   426→  const token = useAuthStore((s) => s.token);

   431→  const [joinOpen, setJoinOpen] = useState(false);
   432→
   433→  const [orders, setOrders] = useState<Order[] | null>(null);
   434→  const [searchCode, setSearchCode] = useState("");
   435→
   436→  useEffect(() => {
   437→    fetch("/api/orders?mine=1", { headers: authHeaders(token) })
   438→      .then((r) => r.json())
   439→      .then((d) => setOrders(Array.isArray(d) ? d : []))
   440→      .catch(() => setOrders([]));

   443→  const shown = (orders || []).filter(
   444→    (o) => !searchCode.trim() || o.code.toLowerCase().includes(searchCode.trim().toLowerCase())
   445→  );
   446→  const active = shown.filter((o) => ["PENDING", "CONFIRMED", "PREPARING", "DELIVERING"].includes(o.status));
   447→  const past = shown.filter((o) => ["DELIVERED", "CANCELLED"].includes(o.status));
   448→
   449→  return (
   450→    <div className="max-w-2xl mx-auto px-4 pb-40 pt-4">
